# GPTBot: OpenAI's model-training data crawler (separate from OAI-SearchBot;
# does not affect ChatGPT search visibility). Blocked by default: the
# dictionary/AI-explanation content is part of this site's product value and
# revenue model, so blocking training-data scraping is the safer, reversible
# default. This has no effect on search-answer visibility, which is handled
# by OAI-SearchBot above.
User-agent: GPTBot
Disallow: /

# ClaudeBot: Anthropic's model-training crawler (separate from Claude's
# user-triggered fetches, e.g. when a user pastes a URL into Claude).
# Same reasoning as GPTBot: blocked by default to protect training-data use
# of original content; reversible if the business decision changes later.
User-agent: ClaudeBot
Disallow: /

# Google-Extended: controls use of this site's content to train
# Gemini/Bard/Vertex AI models. Does NOT affect regular Googlebot search
# crawling/indexing, which stays governed by `User-agent: *` above.
# Same training-data reasoning as GPTBot/ClaudeBot: blocked by default.
User-agent: Google-Extended
Disallow: /
